    CPC_dem_riot_change_weekly = {
		allowed = { tag = CHI }
		icon = GFX_decision_revolt
		activation = {
			CPC = { has_country_flag = CPC_admit_dem_riot }
		}
		available = {
			hidden_trigger = {
				NOT = { CPC = { has_country_flag = CPC_admit_dem_riot } }
			}
		}
		cancel_trigger = {
			hidden_trigger = {
				NOT = { CPC = { has_country_flag = CPC_admit_dem_riot } }
			}
		}
		is_good = no
		days_mission_timeout = 5
		timeout_effect = {
            custom_effect_tooltip = {
			if = {
				limit = { is_ai = yes }
				if = { 
					limit = { check_variable = { CHI_dem_riot_selected^0 = 2 } }
					set_variable = { CHI_dem_riot_selected^0 = 1 }
					set_variable = { CHI_dem_riot_selected^1 = 2 }
				}
				else_if = { 
					limit = { check_variable = { CHI_dem_riot_selected^1 = 2 } }
					set_variable = { CHI_dem_riot_selected^1 = 1 }
					set_variable = { CHI_dem_riot_selected^2 = 2 }
				}
				else_if = { 
					limit = { check_variable = { CHI_dem_riot_selected^2 = 2 } }
					set_variable = { CHI_dem_riot_selected^2 = 1 }
					set_variable = { CHI_dem_riot_selected^0 = 2 }
				}
			}
			set_temp_variable = { TER_dem_power_tmp = -2 }
			set_temp_variable = { TER_dem_radical_tmp = 3 }
			if = {
				limit = { BJC = { exists = yes } }
				set_temp_variable = { target = BJC }
				CHI_dem_riot_power_change = yes
				CHI_dem_riot_radical_change = yes
			}
			if = {
				limit = { WHC = { exists = yes } }
				set_temp_variable = { target = WHC }
				CHI_dem_riot_power_change = yes
				CHI_dem_riot_radical_change = yes
			}
			if = {
				limit = { LZD = { exists = yes } }
				set_temp_variable = { target = LZD }
				CHI_dem_riot_power_change = yes
				CHI_dem_riot_radical_change = yes
			}
			CHI_dem_riot_weekly_effect = yes
		}
        activate_mission = CHI_dem_riot_change_weekly
        }
        
  	}
