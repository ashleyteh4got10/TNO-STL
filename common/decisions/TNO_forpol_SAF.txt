	## EQUIPMENT ##

	SAF_WAC_test = {
		allowed = { original_tag = SAF }
		icon = GFX_decision_generic_prepare_civil_war
		priority = 28

		visible = {
			TNO_fopo_wac_selected = yes
		}

		custom_cost_trigger = {
			has_political_power > 14.99
			command_power > 4.99
		}
		custom_cost_text = decision_cost_5_cp_15_pp

		ai_will_do = {
			factor = 1
			modifier = {
				add = 10
				check_variable = { global.WAC_escalation > 4 }
			}
		}

		days_remove = 21
		days_re_enable = 30
		complete_effect = {
			hidden_effect = { add_command_power = -5 add_political_power = -15 }
			set_country_flag = WAC_active_shipment
			hidden_effect = {
				add_to_variable = { USA_WAC_war_profiteering = 1 }
				if = { # war profiteering event
					limit = { check_variable = { USA_WAC_war_profiteering = 7 } }
					country_event = { id = west_african_crisis.51 days = 2 }
				}
			}
		}
		cancel_effect = {
			clr_country_flag = WAC_active_shipment
		}
		remove_effect = {
			clr_country_flag = WAC_active_shipment
			send_equipment = {
				type = infantry_equipment_2
				amount = 750
				target = FROM
			}
			send_equipment = {
				type = support_equipment
				amount = 300
				target = FROM
			}
			send_equipment = {
				type = artillery_equipment_2
				amount = 100
				target = FROM
			}
			send_equipment = {
				type = anti_tank_equipment_2
				amount = 40
				target = FROM
			}
		}
	}
