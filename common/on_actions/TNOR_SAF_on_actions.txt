on_actions = {
	# FROM = winner
	# ROOT = country being capitulated/loser
	on_capitulation = {
		effect = {
			if = {
				limit = {
					FROM = { is_2GCW_german_nation = yes }
					ROOT = { is_2GCW_german_nation = yes }
				}
				ROOT = {
					transfer_navy = { target = FROM }
				}
			}
		}
	}
	on_startup = {
		effect = {
			SAF = {
			TNOR_init_intro_sequence = yes
			}
		}
	}
}
