                    }

                        annex_country = {
                            target = ROOT
                            transfer_troops = no
                        }
                    }

                        limit = {
                            NOT = {
                                country_exists = POL
                                country_exists = UKR

                                        is_core_of = DST

                    ROOT = {

                            }
                            GER = {
                                set_autonomy = {
                                    target = ROOT
                                    autonomy_state = autonomy_free
                                }
                            }
                        }
                    }
                    set_global_flag = skip_default_capitulation
                }
