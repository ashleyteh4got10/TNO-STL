on_actions = {
    on_startup = {
        effect = {	
			if = {
				limit = {
					has_game_rule = {
						rule = SWRWTER_APRIL_FOOLS_2026
						option = SWRWTER_APRIL_FOOLS_ENABLED
					}
				}
				every_country = {
                limit = {
                    is_ai = no
                }
                country_event = {
                    id = STL_april_fools.1
                    days = 14
                }
            }
			}
			if = {
				every_country = {
					limit = {
						is_ai = no
					}
					country_event = {
						id = STL_support.98
					}
				}
			}

        }
    }

	on_capitulation = {
		effect = {
			if = {
				limit = {
					FROM = {
						original_tag = CPC
					}
					ROOT = {
						original_tag = GER
					}
				}
				CPC ={ white_peace = GER }
			}
			if = {
				limit = {
					FROM = {
						original_tag = MGI
					}
					ROOT = {
						original_tag = USA
					}
				}
				every_country = {
					limit = {
						is_in_faction_with = USA
					}
					MGI = {
						annex_country = {
							target = PREV
							transfer_troops = no
						}
					}
				}
				set_global_flag = skip_default_capitulation
			}
			else_if = {
				limit = {
					FROM = {
						original_tag = MGI
					}
					ROOT = {
						original_tag = CAO
					}
				}
				every_country = {
					limit = {
						is_in_faction_with = CAO
					}
					MGI = {
						annex_country = {
							target = PREV
							transfer_troops = no
						}
					}
				}
				set_global_flag = skip_default_capitulation
			}
			else_if = {
				limit = {
					FROM = {
						original_tag = MGI
					}
					ROOT = {
						original_tag = JAP
					}
				}
				every_country = {
					limit = {
						is_in_faction_with = JAP
					}
					MGI = {
						annex_country = {
							target = PREV
							transfer_troops = no
						}
					}
				}
				set_global_flag = skip_default_capitulation
			}
			else_if = {
				limit = {
					FROM = {
						original_tag = MGI
					}
					ROOT = {
						original_tag = GER
					}
				}
				every_country = {
					limit = {
						is_in_faction_with = GER
					}
					MGI = {
						annex_country = {
							target = PREV
							transfer_troops = no
						}
					}
				}
				set_global_flag = skip_default_capitulation
			}
		}
	}

	on_weekly_MGI = {
		effect = {
			every_country = {
					limit = {
						has_war_with = MGI
						NOT = {
							has_idea = MGI_Magical_Factor
						}
					}
					add_ideas = MGI_Magical_Factor
				}
		}
	}

    on_monthly_MGI = { # Declare War Thingy
		effect = {
			if = {
				limit = {
					OMS = { exists = no }
					MGI = {
						exists = yes
						NOT = {
							has_country_flag = MGI_TGWT_started
						}
					}
				}
			}
			if = {
				limit = {
					OMS = { exists = no }
					MGI = { 
						exists = yes
						has_country_flag = MGI_TGWT_started
					}

				}

				every_country = {
					limit = {
						has_war_with = MGI
						NOT = {
							has_idea = MGI_Magical_Factor
						}
					}
					add_ideas = MGI_Magical_Factor
				}

				if = {
					limit = {
						USA = {
							has_war_with = MGI
						}
					}
					MGI = {
						load_oob = MGI_radiosilence_USA
					}
				}
				if = {
					limit = {
						ENG = {
							has_war_with = MGI
						}
					}
					MGI = {
						load_oob = MGI_radiosilence_UK
					}
				}
				if = {
					limit = {
						JAP = {
							has_war_with = MGI
						}
					}
					MGI = {
						load_oob = MGI_radiosilence_JAP
					}
				}
			}
			MGI = {
				every_controlled_state = {
					limit = { 
						NOT = {
							is_core_of = MGI
						}
					 }
					MGI = { add_state_core = PREV }
				}
			}
		}
	}
	on_weekly_LAO = {
		effect = {
			every_country = {
					limit = {
						has_war_with = LAO
						LAO = {
							has_cosmetic_tag =  LAO_LPDR
						}
						NOT = {
							CPC = {
								OR ={
									has_war_with = JAP
									has_war_with = CHI
								}
						    }
					    }
					every_country = {
						white_peace = LAO
					}
				}
			}
		}
	}
}
