## possible - filters out achivements on game start so we dont have to check everything all the time
## happened - checks if achivement happened

##### BITTER PLUM, COLD FRAGRANCE #####
unique_id = LAR_BP_achievements

LAR_BP_expel_the_japaneses = {
	possible = {
		has_start_date < 1962.01.02
		tag = CHI
	}

	happened = {
		custom_trigger_tooltip = {
			# 西北战争完胜
			has_country_flag = LAR_CHI_ACHIEVE_NORTHWESTWAR_WIN
			tooltip = LAR_BP_expel_the_japaneses_tooltip
		}
	}
}

LAR_BP_bauhinia_evergreen = {
	possible = {
		has_start_date < 1962.01.02
		tag = CHI
	}

	happened = {
		custom_trigger_tooltip = {
			# 检测中国与广东是否处于同一阵营
			GNG = {
				is_in_faction_with = CHI
				NOT = {
					is_in_faction_with = JAP
				}
			}
			tooltip = LAR_BP_bauhinia_evergreen_tooltip
		}
	}
}
