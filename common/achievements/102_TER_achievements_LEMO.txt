# possible - filters out achivements on game start so we dont have to check everything all the
#time
## happened - checks if achivement happened
unique_id = 102_TER_achievements_LEMO
##### CPC #####
TER_LEMO_1_good_bureaucrat = {
	possible = {
		has_start_date < 1962.01.02
		tag = CPC
	}
	happened = {
		custom_trigger_tooltip = {
			check_variable = { CPC_RBM_level > 10 }
			NOT = { has_country_flag = CPC_baby_bus_flag }
			tooltip = TER_good_bureaucrat_achievement
		}
	}
}
