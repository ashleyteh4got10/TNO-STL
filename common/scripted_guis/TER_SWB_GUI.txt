							AND = { NOT = { XIK = { has_country_flag = SWB_focus_begins } }
									XIK = { exists = yes } }
							AND = { XIK = { exists = no } }
						}
					}
					set_variable = { SWB_focus_state^3 = 2 }
					XIK = { set_country_flag = SWB_focus_begins }
					set_country_flag = SWB_focus_begins
					add_political_power = -20

				SWB_draw_the_graph = yes
			}
