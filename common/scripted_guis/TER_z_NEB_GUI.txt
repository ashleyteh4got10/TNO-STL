				}
				set_variable = { NEB_mission = 1 }
				set_variable = { NEB_mission_frame = 1 }
			}
			NEB_mission_2_click = {
				if = {
					limit = {
						NOT = {
							has_country_flag = NEB_mission_201_complete

						}
					}
					set_country_flag = NEB_mission_active
				}
				NEB_mission_2_support = yes
			}
			NEB_mission_3_click = {

				}
