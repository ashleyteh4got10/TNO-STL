	TER_SCB_info_main_GUI = {
		context_type = player_context
		parent_window_name = "CPC_main_GUI"
		window_name = "TER_SCB_info_main_GUI"
		visible = {
			has_country_flag = CPC_main_GUI_open
			check_variable = { CPC_GUI_tab_frames^4 = 2 }
			check_variable = { CPC_GUI_tab_issue^0 = 2 }
			check_variable = { SCB_target_country = QYC }
		}

		triggers = {
			SCB_page_minus_click_enabled = { check_variable = { Huanan_page_num > 1 } }
			SCB_page_add_click_enabled = { check_variable = { Huanan_page_num < TER_Huanan_Bureau_text_cap } }
			SCB_trade_btn_admit_click_enabled = {
				OR = { AND = { check_variable = { SCB_trade_num_i < 4 }
							   check_variable = { QYC.SCB_QYC_light_amount > 0 } }
					   AND = { check_variable = { SCB_trade_num_i > 3 }
					  		   check_variable = { QYC.SCB_QYC_heavy_amount > 0 } } }
			}
			TER_SCB_btn_sp_trade_chosen_visible = {
				check_variable = { TER_SCB_btn_sp_trade_state = 2 }
			}
			TER_SCB_btn_sp_trade_click_enabled = {
				check_variable = { SCB_trade_burden_value > 89.99 }
				check_variable = { SCB_trade_burden_value < 100 }
			}
		}

		dynamic_lists = {
			SCB_trade_list = {
				array = SCB_trade_num
				index = SCB_trade_num_i
				entry_container = SCB_trade_list_ctn
			}
		}

		properties = {
			SCB_Huanan_memorandum_bg_portrait_left = { image = "[SCB_Huanan_memorandum_bg_portrait_left]" }
			SCB_Huanan_memorandum_bg_portrait_right = { image = "[SCB_Huanan_memorandum_bg_portrait_right]" }
			SCB_trade_btn_admit = { frame = SCB_trade_num^SCB_trade_num_i }
			TER_SCB_btn_sp_trade = { frame = TER_SCB_btn_sp_trade_state }
		}

		effects = {
			SCB_page_minus_click = { 
				subtract_from_variable = { Huanan_page_num = 1 }
				TER_SCB_visual_update = yes
			}
			SCB_page_add_click = { 
				add_to_variable = { Huanan_page_num = 1 }
				TER_SCB_visual_update = yes
			}
			SCB_trade_btn_admit_click = { 
				if = { 
					limit = { check_variable = { SCB_trade_num^SCB_trade_num_i = 1 } }
					set_variable = { SCB_trade_num^SCB_trade_num_i = 2 }
					if = {
						limit = { check_variable = { SCB_trade_num_i < 4 } }
						add_to_variable = { SCB_admit_amount_l = 1 }
					}
					else = {
						add_to_variable = { SCB_admit_amount_h = 1 }
					}
				}
				else = { 
					set_variable = { SCB_trade_num^SCB_trade_num_i = 1 }
					if = {
						limit = { check_variable = { SCB_trade_num_i < 4 } }
						subtract_from_variable = { SCB_admit_amount_l = 1 }
					}
					else = {
						subtract_from_variable = { SCB_admit_amount_h = 1 }
					} 
				}
				TER_SCB_weapon_distribution_effect = yes
				TER_SCB_visual_update = yes
			}
			TER_SCB_btn_sp_trade_click = {
				if = {
					limit = { check_variable = { TER_SCB_btn_sp_trade_state = 1 } }
					set_variable = { TER_SCB_btn_sp_trade_state = 2 }
					set_variable = { SCB_trade_sp_burden = 100 }
					subtract_from_variable = { SCB_trade_sp_burden = SCB_trade_burden_value }
					set_variable = { SCB_trade_burden_value = 100 }
					QYC = {
						add_to_variable = { SCB_QYC_monthly_population = 0.02 }
						add_to_variable = { SCB_QYC_stability_factor = 0.02 }
						add_to_variable = { SCB_QYC_production_factory_efficiency_gain_factor = 0.02 }
						add_to_variable = { SCB_QYC_industrial_capacity_factory = 0.02 }
						add_to_variable = { SCB_QYC_production_factory_start_efficiency_factor = 0.02 }
						force_update_dynamic_modifier = yes
					}
					set_variable = { SCB_CPC_production_factory_efficiency_gain_factor = QYC.SCB_QYC_production_factory_efficiency_gain_factor }
					set_variable = { SCB_CPC_industrial_capacity_factory = QYC.SCB_QYC_industrial_capacity_factory }
					set_variable = { SCB_CPC_production_factory_start_efficiency_factor = QYC.SCB_QYC_production_factory_start_efficiency_factor }
					multiply_variable = { SCB_CPC_production_factory_efficiency_gain_factor = 0.2 }
					multiply_variable = { SCB_CPC_industrial_capacity_factory = 0.2 }
					multiply_variable = { SCB_CPC_production_factory_start_efficiency_factor = 0.2 }
					force_update_dynamic_modifier = yes
				}
				else = {
					set_variable = { TER_SCB_btn_sp_trade_state = 2 }
				}
				TER_SCB_visual_update = yes
			}
		}
	}
