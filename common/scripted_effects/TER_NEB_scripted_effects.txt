NEB_support_selected = {
	set_variable = { NEB_temp_province = NEB_province }
	set_variable = { NEB_temp_support = NEB_support_type }
	set_variable = { NEB_import_support^NEB_temp_province = NEB_temp_support }
	if = {
		limit = {
			check_variable = { NEB_support_1_num > 0 }
		}
		subtract_from_variable = { NEB_res_fac_true = NEB_support_1_num }
	}
	if = {
		limit = {
			check_variable = { NEB_support_2_num > 0 }
		}
		subtract_from_variable = { NEB_education_rate = NEB_support_2_num }
	}
	if = {
		limit = {
			check_variable = { NEB_support_3_num > 0 }
		}
		subtract_from_variable = { NEB_infantry_sum = NEB_support_3_num }
		subtract_from_variable = { NEB_support_sum = NEB_support_3_num }
		subtract_from_variable = { NEB_artillery_sum = NEB_support_3_num }

		limit = {
			check_variable = { NEB_support_4_num > 0 }
		}
		subtract_from_variable = { NEB_industrial_equip = NEB_support_4_num }
	}

		limit = {
			check_variable = { NEB_support_5_num > 0 }
		}
		subtract_from_variable = { NEB_healthcare_true = NEB_support_5_num }
	}
	set_variable = { NEB_support_1_num = 0 }
	set_variable = { NEB_support_2_num = 0 }
	set_variable = { NEB_support_3_num = 0 }
	set_variable = { NEB_support_4_num = 0 }
	set_variable = { NEB_support_5_num = 0 }
	for_each_loop = {
		array = NEB_import_support
		index = num
		value = support
		if = {
			limit = {
				check_variable = { support = 1 }
			}
			add_to_variable = { NEB_support_1_num = 1 }
		}
		if = {
			limit = {
				check_variable = { support = 2 }
			}
			add_to_variable = { NEB_support_2_num = 1 }
		}
		if = {
			limit = {
				check_variable = { support = 3 }
			}
			add_to_variable = { NEB_support_3_num = 1 }
		}
		if = {
			limit = {
				check_variable = { support = 4 }
			}
			add_to_variable = { NEB_support_4_num = 1 }
		}
		if = {
			limit = {
				check_variable = { support = 5 }
			}
			add_to_variable = { NEB_support_5_num = 1 }
		}
	}
	multiply_variable = { NEB_support_1_num = 0.11 }
	multiply_variable = { NEB_support_2_num = 0.12 }
	multiply_variable = { NEB_support_3_num = 21 }
	multiply_variable = { NEB_support_4_num = 0.09 }
	multiply_variable = { NEB_support_5_num = 0.10 }
	clamp_variable = {
		var = NEB_support_1_num
		min = 0
		max = 1.5
	}
	clamp_variable = {
		var = NEB_support_2_num
		min = 0
		max = 1.5
	}
	clamp_variable = {
		var = NEB_support_3_num
		min = 0
		max = 105
	}
	clamp_variable = {

		min = 0
		max = 1.5

	clamp_variable = {

		min = 0
		max = 1.5
	}
	add_to_variable = { NEB_res_fac_true = NEB_support_1_num }
	add_to_variable = { NEB_education_rate = NEB_support_2_num }
	add_to_variable = { NEB_infantry_sum = NEB_support_3_num }
