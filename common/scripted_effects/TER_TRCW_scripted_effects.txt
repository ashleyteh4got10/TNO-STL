TRCW_greater_china_provinces_control_change = {
	update_TRCW_map = yes
	CPC = { clear_array = TRCW_controller_status }
	for_each_scope_loop = {
		array = global.greater_china_states
		CPC = { set_temp_variable = { i = PREV.china_state_ref } }
		if = {
			limit = {
				OR = { 
					is_controlled_by = CPC
					controller = { is_in_faction_with = CPC } 
				}
			}
			CPC = { add_to_temp_variable = { TRCW_controller_status_CPC^i = 1 } }
		}
		else_if = {
			limit = {
				OR = { 
					is_controlled_by = CHI
					controller = { is_in_faction_with = CHI } 
				}
			}
			CPC = { add_to_temp_variable = { TRCW_controller_status_JAP^i = 1 } }
		}
		else_if = {
			limit = {
				OR = { 
					is_controlled_by = JAP
					controller = { is_in_faction_with = JAP } 
				}
			}
			CPC = { add_to_temp_variable = { TRCW_controller_status_JAP^i = 1 } }
		}
	}
	for_each_scope_loop = {
		array = global.greater_china_provinces
		clear_variable = TRCW_controller_status
		CPC = { set_temp_variable = { i = PREV.china_state_ref } }
		if = {
			limit = { CPC = { check_variable = { TRCW_controller_status_CPC^i = 0 } } }
			set_variable = { TRCW_controller_status = 3 }
			CPC = { add_to_array = { TRCW_controller_status = 3 } }
		}
		else_if = {
			limit = { CPC = { check_variable = { TRCW_controller_status_JAP^i = 0 } } }
			set_variable = { TRCW_controller_status = 1 }
			CPC = { add_to_array = { TRCW_controller_status = 1 } }
		}
		else_if = {
			limit = { 
				NOT = { CPC = { check_variable = { TRCW_controller_status_JAP^i = 0 } } }
				NOT = { CPC = { check_variable = { TRCW_controller_status_CPC^i = 0 } } }
			}
			set_variable = { TRCW_controller_status = 2 }
			CPC = { add_to_array = { TRCW_controller_status = 2 } }
		}
		else_if = {
			limit = { controller = { is_russian_nation = yes } }
			set_variable = { TRCW_controller_status = 4 }
			CPC = { add_to_array = { TRCW_controller_status = 4 } }
		}
	}
	every_state = {
		limit = { OR = { check_variable = { china_state_ref = 37 }
						 check_variable = { china_state_ref = 53 } } }
		set_variable = { TRCW_controller_status = 0 }
		CPC = { set_variable = { TRCW_controller_status^37 = 0 } }
		CPC = { set_variable = { TRCW_controller_status^53 = 0 } }
	}
	update_TRCW_map = yes
}
