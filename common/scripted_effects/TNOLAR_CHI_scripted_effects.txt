CHI_MND_faction_influences_change = { #使用此方法计算出的数值可能会出现1-5左右的误差
	#下方是三个临时变量，在应用这个效果之前，设定你要变化的数值
	#民族派： temp_CHI_MND_GUI_nationalists_influences_change
	#中央派： temp_CHI_MND_GUI_conservatives_influences_change
	#华北派： temp_CHI_MND_GUI_japanophiles_influences_change

	#应用三个变量的临时调整
	add_to_variable = { CHI_MND_GUI_nationalists_influences_var = temp_CHI_MND_GUI_nationalists_influences_change }
	add_to_variable = { CHI_MND_GUI_conservatives_influences_var = temp_CHI_MND_GUI_conservatives_influences_change }
	add_to_variable = { CHI_MND_GUI_japanophiles_influences_var = temp_CHI_MND_GUI_japanophiles_influences_change }

	#保证不会出现负数
	clamp_variable = {
		var = CHI_MND_GUI_nationalists_influences_var
		min = 0
	}
	clamp_variable = {
		var = CHI_MND_GUI_conservatives_influences_var
		min = 0
	}
	clamp_variable = {
		var = CHI_MND_GUI_japanophiles_influences_var
		min = 0
	}

	#计算调整后三个变量的和
	set_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_nationalists_influences_var }
	add_to_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_conservatives_influences_var }
	add_to_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_japanophiles_influences_var }

	#对调整后的数值进行归一化处理
	divide_temp_variable = { temp_CHI_MND_GUI_influences_total = 100 }
	divide_variable = { CHI_MND_GUI_nationalists_influences_var = temp_CHI_MND_GUI_influences_total }
	divide_variable = { CHI_MND_GUI_conservatives_influences_var = temp_CHI_MND_GUI_influences_total }
	divide_variable = { CHI_MND_GUI_japanophiles_influences_var = temp_CHI_MND_GUI_influences_total }
	round_variable = CHI_MND_GUI_nationalists_influences_var
	round_variable = CHI_MND_GUI_conservatives_influences_var
	round_variable = CHI_MND_GUI_japanophiles_influences_var
	

	#再次计算归一化后三个数值的和
	set_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_nationalists_influences_var }
	add_to_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_conservatives_influences_var }
	add_to_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_japanophiles_influences_var }

	if = {
		limit = { #如果数值之和正好等于100
			check_variable = { temp_CHI_MND_GUI_influences_total = 100 }
		}
		#do nothing
	}
	else = {	#对误差进行修正，修正数值偏向民族派
		subtract_from_temp_variable = { temp_CHI_MND_GUI_influences_total = 100 }
		if = {
			limit = { #数值之和大于或小于100，将它减去100后仍为正数
				check_variable = { temp_CHI_MND_GUI_influences_total > 0 }
			}
			subtract_from_variable = { CHI_MND_GUI_japanophiles_influences_var = temp_CHI_MND_GUI_influences_total }
		}

		else_if = {
			limit = { #数值之和大于或小于100，将它减去100后为负数
				check_variable = { temp_CHI_MND_GUI_influences_total < 0 }
			}
			subtract_from_variable = { CHI_MND_GUI_nationalists_influences_var = temp_CHI_MND_GUI_influences_total }
		}
	}

	#再次检查三个变量之和是否正确，之后记得注释掉
	set_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_nationalists_influences_var }
	add_to_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_conservatives_influences_var }
	add_to_temp_variable = { temp_CHI_MND_GUI_influences_total = CHI_MND_GUI_japanophiles_influences_var }
	if = {
		limit = {
			NOT = {
				check_variable = { temp_CHI_MND_GUI_influences_total = 100 }
			}
		}
		log = "[GetDateText]: [Root.GetName]: ERROR:MND GUI influences Calculation result incorrect"
	}
	
	custom_effect_tooltip = CHI_MND_faction_influences_change_tt
	CHI_MND_faction_value_change = yes
}
