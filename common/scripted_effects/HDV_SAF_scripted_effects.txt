## give this scripted effect a postive number for SAF_society_progressivism_temp please!
## moves progressivism towards 100 (status quo) by the amount of points specified in the temp variable 
## if the value of the temp variable would overshoot 100, (i.e progressivism is 105 and you give it 15)
## the change will be modified to not overshoot (i.e the change would become 5 instead of 15 to exactly hit 100)
SAF_Society_progressivism_middle_change = {
	# calculate diff between current and 100
	set_temp_variable = { SAF_diff_from_status_quo = SAF_society_progressivism }
	subtract_from_temp_variable = { SAF_diff_from_status_quo = 100 }

	set_temp_variable = { SAF_diff_from_status_quo_abs = SAF_diff_from_status_quo }
	if = { 
		limit = {
			check_variable = { SAF_diff_from_status_quo < 0 }
		}
		
		multiply_temp_variable = { SAF_diff_from_status_quo_abs = -1 }
	}

	if = {
		limit = {
			check_variable = { SAF_society_progressivism_temp > SAF_diff_from_status_quo_abs }
		}
		set_temp_variable = { SAF_society_progressivism_temp = SAF_diff_from_status_quo_abs }
	}

	if = {
		limit = {
			check_variable = { SAF_society_progressivism > 100 }
		}
		multiply_temp_variable = { SAF_society_progressivism_temp = -1 }
	}

	custom_effect_tooltip = SAF_society_status_quo_tt

	hidden_effect = {
		SAF_Society_progressivism_change = yes
	}
}

SAF_update_dev_progress = {
	for_each_loop = {
		array = SAF_actual_ddp
		index = i
		if = {
			limit = {
				check_variable = { SAF_daily_dev_progress^i > 0 }
			}
			set_variable = { SAF_actual_ddp^i = SAF_daily_dev_progress^i }
			divide_variable = { SAF_actual_ddp^i = SAF_development^i }
		}
		else_if = {
			limit = {
				check_variable = { SAF_daily_dev_progress^i < 0 }
			}
			set_temp_variable = { SAF_minus_10_temp = 10 }
			subtract_from_temp_variable = { SAF_minus_10_temp = SAF_development^i }
			set_variable = { SAF_actual_ddp^i = SAF_daily_dev_progress^i }
			divide_variable = { SAF_actual_ddp^i = SAF_minus_10_temp }
		}
		if = {
			limit = {
				NOT = {
					check_variable = { i = 16 }
				}
				check_variable = { i > 13 }
			}
			if = {
				limit = {
					check_variable = { SAF_daily_dev_progress^i > 0 }
				} 
				multiply_variable = { SAF_actual_ddp^i = 3 }
				else = {
					divide_variable = { SAF_actual_ddp^i = 3 }
				}
			}
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { i = 1 }
					check_variable = { i = 4 }
					check_variable = { i = 5 }
					check_variable = { i = 6 }
					check_variable = { i = 9 }
					check_variable = { i = 10 }
					check_variable = { i = 12 }
				}
			}
			if = {
				limit = {
					check_variable = { SAF_daily_dev_progress^i > 0 }
				} 
				multiply_variable = { SAF_actual_ddp^i = 2 }
				else = {
					divide_variable = { SAF_actual_ddp^i = 2 }
				}
			}
