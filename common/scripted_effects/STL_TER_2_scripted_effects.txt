TER_recover_speed_buff_effect = {
	set_variable = { recover_speed = 100 }
	add_to_temp_array = { CPC_region_array = token:NWB }
	add_to_temp_array = { CPC_region_array = token:SWB }
	add_to_temp_array = { CPC_region_array = token:NCB }
	add_to_temp_array = { CPC_region_array = token:SCB }
	add_to_temp_array = { CPC_region_array = token:NEB }
	add_to_temp_array = { CPC_region_array = token:MON }
	add_to_temp_array = { CPC_region_array = token:XJB }
	add_to_temp_array = { CPC_region_array = token:XGZ }
	add_to_temp_array = { CPC_region_array = token:NFE }
	add_to_temp_array = { CPC_region_array = token:center }
	for_each_loop = {
		array = CPC_region_array
		value = v
		meta_effect = {
			text = {
				if = {
					limit = { check_variable = { PRC_[REGION]_PP_provided < PRC_PP_[REGION]_needed } }
					if = {
						limit = {
							set_temp_variable = { PP_compar_var = PRC_PP_[REGION]_needed }
							divide_temp_variable = { PP_compar_var = 2 }
							check_variable = { PRC_[REGION]_PP_provided > PP_compar_var }
						}
						subtract_from_variable = { recover_speed = 2 }
					}
					else_if = {
						limit = {
							set_temp_variable = { PP_compar_var = PRC_PP_[REGION]_needed }
							multiply_temp_variable = { PP_compar_var = 0.1 }
							check_variable = { PRC_[REGION]_PP_provided < PP_compar_var }
						}
						set_variable = { recover_speed = 0 }
					}
					else = {
						subtract_from_variable = { recover_speed = 8 }
					}
				}
			}
			REGION = "[?v.GetTokenKey]"
		}
	}
	add_to_variable = { recover_speed = recover_speed_temp }
	clamp_variable = { var = recover_speed min = 0 max = 100 }
}
