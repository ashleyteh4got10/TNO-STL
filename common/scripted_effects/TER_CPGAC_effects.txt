	divide_variable = { CPGAC_Y_attitude_in_party^0 = CPGAC_satisfaction^1 }
	divide_variable = { CPGAC_Y_attitude_in_party^1 = CPGAC_satisfaction^1 }

	divide_variable = { CPGAC_Y_attitude_in_party^3 = CPGAC_satisfaction^1 }
	
	for_loop_effect = {
		start = 0
		end = 4
		value = i
		multiply_temp_variable = { attitude_change_Y_in_party_tmp^i = CPGAC_satisfaction^1 }
	}

	add_to_variable = { CPGAC_Y_attitude_in_party^0 = attitude_change_Y_in_party_tmp^0 }
	add_to_variable = { CPGAC_Y_attitude_in_party^1 = attitude_change_Y_in_party_tmp^1 }
	add_to_variable = { CPGAC_Y_attitude_in_party^2 = attitude_change_Y_in_party_tmp^2 }
