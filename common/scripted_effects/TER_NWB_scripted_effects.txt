TER_NWB_complete_task = {
	var:NWB_executing_task_state = { 
		add_to_variable = { NWB_task_level^CPC.NWB_executing_task_type = 1 }
		clamp_variable = {
			var = NWB_task_level^CPC.NWB_executing_task_type
			min = 0
			max = 3
		}
		if = {
			limit = { check_variable = { NWB_task_level^CPC.NWB_executing_task_type = 3 } }
			add_to_variable = { NWB_unity_level = 1 }
			clamp_variable = {
				var = NWB_unity_level
				min = 0
				max = 6
			}
			TER_NWB_task_complete_give_effect = yes
		}
		set_variable = { NWB_btn_visual_ctrl^0 = 1 }
		set_variable = { NWB_btn_visual_ctrl^1 = 1 }
		set_variable = { NWB_btn_visual_ctrl^2 = 1 }
		set_variable = { NWB_btn_visual_ctrl^3 = 1 }
		set_variable = { NWB_btn_visual_ctrl^4 = 1 }
		set_variable = { NWB_btn_visual_ctrl^5 = 1 }
		set_variable = { NWB_btn_visual_ctrl^6 = 1 }
	}
	clear_variable = NWB_executing_task_state
	clear_variable = NWB_executing_task_type
	clr_country_flag = NWB_activate_task
	CPC = { set_variable = { NWB_unity_level_sum = 0 } }
	every_state = {
		limit = { 
			check_variable = { china_northwest_ref > 0 } 
			check_variable = { NWB_unity_level = 6 }
		}
		CPC = { add_to_variable = { NWB_unity_level_sum = PREV.NWB_unity_level } }
	}
	if = {
		limit = { 
			check_variable = { NWB_unity_level_sum < 90 } 
			NOT = { check_variable = { CPC_hosting_service_NWB_var = 2 } }
		}
		Alert_NWB_enabled_alerts = yes
	}
	else_if = {
		limit = { check_variable = { NWB_unity_level_sum > 89 } }
		Alert_NWB_enabled_remove_alerts = yes
		set_country_flag = CPC_NWB_all_missions_completed
	}
	hidden_effect = {
		TER_NWB_calc_party_members = yes
		TER_NWB_update_visual_effect = yes
	}
}

TER_NWB_task_complete_give_effect = {
	var:NWB_executing_task_state = {
		if = {
			limit = { check_variable = { NWB_unity_level > 1 } }
			var:NWB_executing_task_state = { 
				if = {
					limit = { NOT = { is_core_of = CPC } }
					add_core_of = CPC
					add_building_construction = {
						type = office
						level = 1
					}
				}	
			}
		}
		CPC = {
			if = { 
				limit = { check_variable = { PREV.NWB_task_level^NWB_executing_task_type = 3 } }
				if = {
					limit = { check_variable = { NWB_executing_task_type = 1 } }
					add_to_variable = { agriculture_counter = 5 }
				}
				else_if = {
					limit = { check_variable = { NWB_executing_task_type = 2 } }
					PREV = {
						set_temp_variable = { state_value_multiplier_temp = 1.075 }
						econ_state_value_change_multiply_specified_state = yes
					}
				}
				else_if = {
					limit = { check_variable = { NWB_executing_task_type = 3 } }
					PREV = {
						add_building_construction = {
							type = infrastructure
							level = 1
						}
					}
				}
				else_if = {
					limit = { check_variable = { NWB_executing_task_type = 4 } }
					PREV = {
						add_building_construction = {
							type = schools
							level = 1
						}
					}
				}
				else_if = {
					limit = { check_variable = { NWB_executing_task_type = 5 } }
					PREV = {
						add_building_construction = {
							type = prisons
							level = 1
						}
					}
				}
				else_if = {
					limit = { check_variable = { NWB_executing_task_type = 6 } }
					PREV = {
						add_building_construction = {
							type = hospitals
							level = 1
						}
					}
				}
			}
		}
	}
}

				limit = { CPC = { NOT = { has_country_flag = CPC_unlock_NWB } } }
				set_variable = { NCCPC_PP_NWB_needed = 0 }
			}
			else = {
				set_variable = { NCCPC_PP_NWB_needed = 420 }
				every_state = {
					limit = { 
						OR = {
							check_variable = { china_northwest_ref > 0 }
							check_variable = { china_northwest_ref < 0 }
						}
					}
					set_variable = { NWB_PP_needed = this.state_population_k }
					multiply_variable = { NWB_PP_needed = 0.01 }
					CPC = { add_to_variable = { NCCPC_PP_NWB_needed = PREV.NWB_PP_needed } }
				}
			}
			#西南局所需党员计算
			if = {
				limit = { CPC = { NOT = { has_country_flag = CPC_unlock_SWB } } }
				set_variable = { NCCPC_PP_SWB_needed = 0 }
			}
			else_if = {
				limit = { XIK = { exists = yes } }
				set_variable = { NCCPC_PP_SWB_needed = XIK.max_manpower_k }
				multiply_variable = { NCCPC_PP_SWB_needed = 0.017 }
				every_state = {
					limit = { is_controlled_by = XIK }
					set_state_flag = CPC_this_is_SWB_state
				}
			}
			else_if = {
				limit = { XIK = { exists = no } }
				set_variable = { NCCPC_PP_SWB_needed = 0 }
				every_state = {
					limit = { has_state_flag = CPC_this_is_SWB_state }
					set_variable = { SWB_PP_needed = this.state_population_k }
					multiply_variable = { SWB_PP_needed = 0.017 }
					CPC = { add_to_variable = { NCCPC_PP_SWB_needed = PREV.SWB_PP_needed } }
				}
			}
			#新疆局所需党员计算
			if = {
				limit = { 
					OR = {
						CPC = { has_country_flag = CPC_XJ_mess_up }
						CPC = { NOT = { has_country_flag = CPC_unlock_XJB } } 
					}
				}
				set_variable = { NCCPC_PP_XJB_needed = 0 }
			}
			else = {
				set_variable = { NCCPC_PP_XJB_needed = HTC.max_manpower_k }
				multiply_variable = { NCCPC_PP_XJB_needed = 0.03 }
			}
			#当前党员数
			set_variable = { NCCPC_current_party_pop = 0 }
			set_variable = { NCCPC_current_party_pop_from_shaanxi = 0 }
			set_temp_variable = { NCCPC_PP_tmp_SWB = NCCPC_PP_SWB_needed }
			multiply_temp_variable = { NCCPC_PP_tmp_SWB = 0.6 }
			set_temp_variable = { NCCPC_PP_tmp_XJB = NCCPC_PP_XJB_needed }
			multiply_temp_variable = { NCCPC_PP_tmp_XJB = 0.4 }
			add_to_variable = { NCCPC_current_party_pop = NCCPC_PP_tmp_SWB }
			add_to_variable = { NCCPC_current_party_pop = NCCPC_PP_tmp_XJB }
			every_state = {
				limit = { check_variable = { china_northwest_ref < 0 } }
				set_variable = { this.NCCPC_current_party_pop_in_state = this.state_population_k }
				multiply_variable = { this.NCCPC_current_party_pop_in_state = this.CPC_NWB_party_member_percent }
				CPC = { add_to_variable = { NCCPC_current_party_pop_from_shaanxi = PREV.NCCPC_current_party_pop_in_state } }
			}
			every_state = {
				limit = { 
					OR = {
						check_variable = { china_northwest_ref > 0 }
						check_variable = { china_northwest_ref < 0 }
					}
				}
				set_variable = { this.NCCPC_current_party_pop_in_state = this.state_population_k }
				multiply_variable = { this.NCCPC_current_party_pop_in_state = this.CPC_NWB_party_member_percent }
				CPC = { add_to_variable = { NCCPC_current_party_pop = PREV.NCCPC_current_party_pop_in_state } }	
			}
			#buff放大比例
			if = {
				limit = { 
					has_country_flag = CPC_unlock_NWB
				}
				#西北局比例计算，如果党员数目短缺，禁止西北局执行六大任务
				set_variable = { TER_NWB_PTMB_num_pct = NCCPC_PP_NWB_provided }
				divide_variable = { TER_NWB_PTMB_num_pct = NCCPC_PP_NWB_needed }
				#党代会比例计算，党员总数/理论需求党员总数 - 比例乘算政策之舞理论buff
				set_variable = { TER_NCCPC_PTMB_num_pct = NCCPC_current_party_pop }
				set_temp_variable = { NCCPC_pp_needed_tmp = NCCPC_PP_NWB_needed }
				add_to_temp_variable = { NCCPC_pp_needed_tmp = NCCPC_PP_SWB_needed }
				add_to_temp_variable = { NCCPC_pp_needed_tmp = NCCPC_PP_XJB_needed }
				divide_variable = { TER_NCCPC_PTMB_num_pct = NCCPC_pp_needed_tmp }
			}
			else = {
				set_variable = { TER_NCCPC_PTMB_num_pct = 1 }
				set_variable = { TER_NWB_PTMB_num_pct = 1 }
			}
			if = {
				limit = { 
					has_country_flag = CPC_unlock_SWB
				}
				#西南局比例计算，党员总数/理论需求党员总数 - 比例乘算西南局理论buff
				set_variable = { TER_SWB_PTMB_num_pct = NCCPC_PP_SWB_provided }
				divide_variable = { TER_SWB_PTMB_num_pct = NCCPC_PP_SWB_needed }
			}
			else = {
				set_variable = { TER_SWB_PTMB_num_pct = 1 }
			}
			if = {
				limit = { 
					has_country_flag = CPC_unlock_XJB
					NOT = { has_country_flag = CPC_unlock_SWB }
					NOT = { has_country_flag = CPC_unlock_NWB }
				}
				set_variable = { TER_XJB_PTMB_num_pct = 1 }
			}
			else_if = {
				limit = { NOT = { has_country_flag = CPC_unlock_XJB } }
				set_variable = { TER_XJB_PTMB_num_pct = 1 }
			}
			else_if = {
				limit = { has_country_flag = CPC_unlock_XJB }
				set_variable = { TER_XJB_PTMB_num_pct = NCCPC_PP_XJB_provided }
				divide_variable = { TER_XJB_PTMB_num_pct = NCCPC_PP_XJB_needed }
			}
		}
		TER_NWB_calc_state_modifiers = yes
		TER_CPC_update_ptmb_pct_dynmod = yes
		set_variable = { NCCPC_current_printed_party_pop = NCCPC_current_party_pop }
		divide_variable = { NCCPC_current_printed_party_pop = 1000 }
	}

	set_variable = { NCCPC_PP_XJB_provided = NCCPC_PP_XJB_needed }
	set_variable = { NCCPC_PP_SWB_provided = NCCPC_PP_SWB_needed }
	set_variable = { NCCPC_PP_NWB_provided = NCCPC_PP_NWB_needed }

	set_temp_variable = { NWB_PTMB_num_tmp = NCCPC_current_party_pop }
	subtract_from_temp_variable = { NWB_PTMB_num_tmp = NCCPC_PP_XJB_provided }
	subtract_from_temp_variable = { NWB_PTMB_num_tmp = NCCPC_PP_SWB_provided }
	subtract_from_temp_variable = { NWB_PTMB_num_tmp = NCCPC_PP_NWB_provided }
	
	#求和
	set_temp_variable = { NWB_PTMB_num_needed_sum_tmp = NCCPC_PP_XJB_needed }
	add_to_temp_variable = { NWB_PTMB_num_needed_sum_tmp = NCCPC_PP_SWB_needed }
	add_to_temp_variable = { NWB_PTMB_num_needed_sum_tmp = NCCPC_PP_NWB_needed }

	#占比
	set_temp_variable = { NWB_PTMB_num_XJB_pct_tmp = NCCPC_PP_XJB_needed }
	divide_temp_variable = { NWB_PTMB_num_XJB_pct_tmp = NWB_PTMB_num_needed_sum_tmp }
	set_temp_variable = { NWB_PTMB_num_SWB_pct_tmp = NCCPC_PP_SWB_needed }
	divide_temp_variable = { NWB_PTMB_num_SWB_pct_tmp = NWB_PTMB_num_needed_sum_tmp }
	set_temp_variable = { NWB_PTMB_num_NWB_pct_tmp = NCCPC_PP_NWB_needed }
	divide_temp_variable = { NWB_PTMB_num_NWB_pct_tmp = NWB_PTMB_num_needed_sum_tmp }	

	#分配冗余
	set_temp_variable = { NCCPC_PP_XJB_redundance = NWB_PTMB_num_tmp }
	set_temp_variable = { NCCPC_PP_SWB_redundance = NWB_PTMB_num_tmp }
	set_temp_variable = { NCCPC_PP_NWB_redundance = NWB_PTMB_num_tmp }
	multiply_temp_variable = { NCCPC_PP_XJB_redundance = NWB_PTMB_num_XJB_pct_tmp }
	multiply_temp_variable = { NCCPC_PP_SWB_redundance = NWB_PTMB_num_SWB_pct_tmp }
	multiply_temp_variable = { NCCPC_PP_NWB_redundance = NWB_PTMB_num_NWB_pct_tmp }

	#加算
	add_to_variable = { NCCPC_PP_XJB_provided = NCCPC_PP_XJB_redundance }
	add_to_variable = { NCCPC_PP_SWB_provided = NCCPC_PP_SWB_redundance }
	add_to_variable = { NCCPC_PP_NWB_provided = NCCPC_PP_NWB_redundance }

	set_temp_variable = { NCCPC_PP_sum_provided = NCCPC_PP_XJB_provided }
	add_to_temp_variable = { NCCPC_PP_sum_provided = NCCPC_PP_SWB_provided }
	add_to_temp_variable = { NCCPC_PP_sum_provided = NCCPC_PP_NWB_provided }
	if = {
		limit = { NOT = { check_variable = { NCCPC_PP_sum_provided = NCCPC_current_party_pop } } }
		subtract_from_temp_variable = { NCCPC_PP_sum_provided = NCCPC_current_party_pop }
		if = {
			limit = {
				has_country_flag = CPC_unlock_NWB
				OR = {
					NOT = { has_country_flag = CPC_unlock_XJB }
					has_country_flag = CPC_XJ_mess_up
				}
			}
			divide_temp_variable = { NCCPC_PP_sum_provided = 2 }
			subtract_from_variable = { NCCPC_PP_SWB_provided = NCCPC_PP_sum_provided }
			subtract_from_variable = { NCCPC_PP_NWB_provided = NCCPC_PP_sum_provided }		  
		}
		else_if = {
			limit = { 
				has_country_flag = CPC_unlock_NWB
				has_country_flag = CPC_unlock_XJB 
			}
			divide_temp_variable = { NCCPC_PP_sum_provided = 3 }
			subtract_from_variable = { NCCPC_PP_SWB_provided = NCCPC_PP_sum_provided }
			subtract_from_variable = { NCCPC_PP_NWB_provided = NCCPC_PP_sum_provided }
			subtract_from_variable = { NCCPC_PP_XJB_provided = NCCPC_PP_sum_provided }
		}
	}

	TER_NWB_calc_party_members = yes
