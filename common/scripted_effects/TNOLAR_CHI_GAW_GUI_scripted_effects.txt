CHI_Update_Great_Asian_War_map_GUI_Korean = {
	set_variable = { Great_Asian_states_map_Calc_JAP = 0 }
	set_variable = { Great_Asian_states_map_Calc_CHI = 0 }
	
	for_each_scope_loop = {
		array = CHI.Great_Asian_states_map_LAR_GAW_Korean
		if = {
			limit = {
				OR = {
					is_controlled_by = JAP
					controller = { is_in_faction_with = JAP }
				}
			}
			CHI = {
				add_to_variable = { Great_Asian_states_map_Calc_JAP = 1 }
			}
		}
		else_if = {
			limit = {
				OR = {
					is_controlled_by = CHI
					controller = { is_in_faction_with = CHI }
				}
			}
			CHI = {
				add_to_variable = { Great_Asian_states_map_Calc_CHI = 1 }
			}
		}
	}

	if = {
		limit = {
			check_variable = { Great_Asian_states_map_Calc_JAP > 0 }
			check_variable = { Great_Asian_states_map_Calc_CHI < 1 }
		}
		set_variable = { CHI_Great_Asian_War_map_GUI_Indication_LAR_GAW_Korean_var = 1 }
		log = "[GetDateText]: [Root.GetName]: Korean calc jap control"
	}
	else_if = {
		limit = {
			check_variable = { Great_Asian_states_map_Calc_JAP > 0 }
			check_variable = { Great_Asian_states_map_Calc_CHI > 0 }
		}
		set_variable = { CHI_Great_Asian_War_map_GUI_Indication_LAR_GAW_Korean_var = 2 }
		log = "[GetDateText]: [Root.GetName]: Korean calc draw"
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					check_variable = { Great_Asian_states_map_Calc_JAP < 1 }
					check_variable = { Great_Asian_states_map_Calc_CHI > 0 }
				}
				AND = {
					check_variable = { Great_Asian_states_map_Calc_JAP = 0 }
					check_variable = { Great_Asian_states_map_Calc_CHI = 0 }
				}
			}
		}
		set_variable = { CHI_Great_Asian_War_map_GUI_Indication_LAR_GAW_Korean_var = 3 }
		log = "[GetDateText]: [Root.GetName]: Korean calc chi control"
	}
}
