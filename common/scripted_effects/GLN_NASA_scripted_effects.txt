GLN_fund_NASA_once_effect = {
	set_temp_variable = { temp_econ_spending_amount_unmodified = temp_econ_spending_amount }
	if = {
		limit = {
			check_variable = {
				var = temp_econ_spending_amount
				value = money_reserves
				compare = less_than_or_equals
			}
		}
		subtract_from_variable = { money_reserves = temp_econ_spending_amount }
		add_to_variable = { USA_GLN_NASA_Funding = temp_econ_spending_amount }
		custom_effect_tooltip = TNO_spend_money_tt
		GLN_NASA_recalculate_budget_left = yes
	}
	else_if = {
		limit = {
			check_variable = { temp_econ_spending_amount_unmodified < 0 }
		}
		set_temp_variable = { money_reserves_temp = temp_econ_spending_amount_unmodified }
		multiply_temp_variable = { money_reserves_temp = -1 }
		econ_money_reserves_change_raw_money = yes
		multiply_temp_variable = { money_reserves_temp = -1 }
		add_to_variable = { USA_GLN_NASA_Funding = temp_econ_spending_amount}
		GLN_NASA_recalculate_budget_left = yes
		
	}
	else = {
		subtract_from_temp_variable = { temp_econ_spending_amount = money_reserves }
		set_variable = { money_reserves = 0 }
		add_to_variable = { national_debt = temp_econ_spending_amount }
		add_to_variable = { USA_GLN_NASA_Funding = temp_econ_spending_amount }
		custom_effect_tooltip = TNO_spend_money_tt
		GLN_NASA_recalculate_budget_left = yes
	}
	update_economy_tab = yes
	
}

GLN_NASA_recalculate_budget_left = {
	set_variable = { USA_GLN_NASA_Budget_Leftover = USA_GLN_NASA_Funding }
	set_temp_variable = { budget_impact = USA_GLN_NASA_Mission_Investment }
	#divide_temp_variable = { budget_impact = 1000 } #Funding is in millions and is directly used in a dynmod so we need to reduce it first
	subtract_from_variable = { USA_GLN_NASA_Budget_Leftover = budget_impact }
	set_variable = { USA_GLN_NASA_Funding_Display = USA_GLN_NASA_Funding }
	multiply_variable = { USA_GLN_NASA_Funding_Display = 1000 }
	set_variable = { USA_GLN_NASA_Funding_Annual_Display = USA_GLN_NASA_Funding_Annual }
	multiply_variable = { USA_GLN_NASA_Funding_Annual_Display = 1000 }
	if = {
		limit = { check_variable = { USA_GLN_NASA_Budget_Leftover < 0 } }
		log = "[GetDateText]: [USA.GetName]: Glenn Negative NASA Budget"
		add_to_variable = { USA_GLN_NASA_Mission_Investment = USA_GLN_NASA_Budget_Leftover }
		set_variable = { USA_GLN_NASA_Budget_Leftover = 0 }
	}
	set_variable = { USA_GLN_NASA_Mission_Investment_Display = USA_GLN_NASA_Mission_Investment }
	multiply_variable = { USA_GLN_NASA_Mission_Investment_Display = 1000 }
}
