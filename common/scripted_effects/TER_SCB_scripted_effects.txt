TER_SCB_monthly_update = {
	QYC = {
		set_variable = { SCB_QYC_monthly_population = 0 }
		set_variable = { SCB_QYC_stability_factor = 0 }
		set_variable = { SCB_QYC_production_factory_efficiency_gain_factor = 0 }
		set_variable = { SCB_QYC_industrial_capacity_factory = 0 }
		set_variable = { SCB_QYC_production_factory_start_efficiency_factor = 0 }
		set_variable = { SCB_QYC_heavy_amount = 0 }
		set_variable = { SCB_QYC_light_amount = 0 }
	}
	#贸易意愿结算
	for_each_scope_loop = {
		array = global.SCB_trade_selectable_country
		if = {
			limit = { check_variable = { SCB_trade_state = 2 }
				      NOT = { original_tag = CPC } }
			randomize_temp_variable = {
				var = SCB_trade_will_value_change
				distribution = binomial
				min = 10
				max = 30
			}
			add_to_variable = { SCB_trade_will_value = SCB_trade_will_value_change }
			clamp_variable = {
				var = SCB_trade_will_value
				min = 0
				max = 100
			}
		}
		else_if = {
			limit = { check_variable = { SCB_trade_state = 1 }
					  NOT = { original_tag = CPC } }
			randomize_temp_variable = {
				var = SCB_trade_will_value_change
				distribution = binomial
				min = -30
				max = -10
			}
			add_to_variable = { SCB_trade_will_value = SCB_trade_will_value_change }
			clamp_variable = {
				var = SCB_trade_will_value
				min = 0
				max = 100
			}
			
		}
	}
	#贸易费用结算
	randomize_temp_variable = {
		var = SCB_trade_cost_seed
		distribution = binomial
		min = 0.001
		max = 0.003
	}
	multiply_temp_variable = { SCB_trade_cost_seed = SCB_start_trade_num }
	set_temp_variable = { debt_temp = SCB_trade_cost_seed }
	econ_national_debt_change = yes
	#警觉度结算判断
	for_each_scope_loop = {
		array = global.SCB_trade_selectable_country
		TER_SCB_JP_alert_seed = yes
		add_to_variable = { CPC.SCB_JP_alert_value = SCB_JP_alert_change }
		if = { 
			limit = { CPC = { check_variable = { CPC_hosting_service_SCB_var = 2 } } }
			subtract_from_variable = { CPC.SCB_JP_alert_value = 2.5	}
		}
	}
	#可革命国家援助效果结算
	VIC = {
		if = {
			limit = { check_variable = { SCB_trade_state = 2 } }
			random_list = {
				34 = { add_to_variable = { TER_SCB_rev_level_unify = 1 } }
				33 = { add_to_variable = { TER_SCB_rev_level_equip = 1 } }
				33 = { add_to_variable = { TER_SCB_rev_level_org = 1 } }
			}
		}
	}
	KPA = {
		if = {
			limit = { check_variable = { SCB_trade_state = 2 } }
			random_list = {
				34 = { add_to_variable = { TER_SCB_rev_level_unify = 1 } }
				33 = { add_to_variable = { TER_SCB_rev_level_equip = 1 } }
				33 = { add_to_variable = { TER_SCB_rev_level_org = 1 } }
			}
		}
	}
	CRC = {
		if = {
			limit = { check_variable = { SCB_trade_state = 2 } }
			random_list = {
				34 = { add_to_variable = { TER_SCB_rev_level_unify = 1 } }
				33 = { add_to_variable = { TER_SCB_rev_level_equip = 1 } }
				33 = { add_to_variable = { TER_SCB_rev_level_org = 1 } }
			}
		}
	}
	#QYC贸易buff结算
	for_each_scope_loop = {
		array = global.SCB_trade_selectable_country
		TER_SCB_monthly_update_trade_seed = yes
	}
	if = {
		limit = { check_variable = { TER_SCB_btn_sp_trade_state = 2 } }
		set_variable = { SCB_trade_burden_value = 100 }
		QYC = {
			add_to_variable = { SCB_QYC_monthly_population = 0.02 }
			add_to_variable = { SCB_QYC_stability_factor = 0.02 }
			add_to_variable = { SCB_QYC_production_factory_efficiency_gain_factor = 0.02 }
			add_to_variable = { SCB_QYC_industrial_capacity_factory = 0.02 }
			add_to_variable = { SCB_QYC_production_factory_start_efficiency_factor = 0.02 }
			force_update_dynamic_modifier = yes
		}
	}
	TER_SCB_weapon_distribution_effect = yes
	QYC = {
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = CPC.SCB_Hainan_trade_equip_amount^0
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = CPC.SCB_Hainan_trade_equip_amount^1
		}
		add_equipment_to_stockpile = {
			type = anti_tank_equipment
			amount = CPC.SCB_Hainan_trade_equip_amount^2
		}
		add_equipment_to_stockpile = {
			type = anti_air_shoulder_equipment
			amount = CPC.SCB_Hainan_trade_equip_amount^3
		}
		add_equipment_to_stockpile = {
			type = artillery_equipment
			amount = CPC.SCB_Hainan_trade_equip_amount^4
		}
		add_equipment_to_stockpile = {
			type = motorized_equipment

	#CPC贸易buff结算
	set_variable = { SCB_CPC_production_factory_efficiency_gain_factor = QYC.SCB_QYC_production_factory_efficiency_gain_factor }
	set_variable = { SCB_CPC_industrial_capacity_factory = QYC.SCB_QYC_industrial_capacity_factory }
	set_variable = { SCB_CPC_production_factory_start_efficiency_factor = QYC.SCB_QYC_production_factory_start_efficiency_factor }

		multiply_variable = { SCB_CPC_industrial_capacity_factory = 0.2 }
		multiply_variable = { SCB_CPC_production_factory_start_efficiency_factor = 0.2 }
	}
	#视觉更新
	set_variable = { SCB_target_country = SCB_pinned_country }
	for_each_scope_loop = {
		array = global.SCB_trade_selectable_country
		clr_country_flag = TER_SCB_trade_GUI_clicked_1
		clr_country_flag = TER_SCB_trade_GUI_clicked_2

	}

TER_SCB_weapon_distribution_effect = {
	set_temp_variable = { SCB_QYC_heavy_amount_temp = QYC.SCB_QYC_heavy_amount }
	set_temp_variable = { SCB_QYC_light_amount_temp = QYC.SCB_QYC_light_amount }
	if = { limit = { check_variable = { SCB_admit_amount_l > 0 } }
		   divide_temp_variable = { SCB_QYC_light_amount_temp = SCB_admit_amount_l } }
	if = { limit = { check_variable = { SCB_admit_amount_h > 0 } }
		   divide_temp_variable = { SCB_QYC_heavy_amount_temp = SCB_admit_amount_h } }
	for_each_loop = {
		array = SCB_Hainan_trade_equip_amount
		index = i
		if = {
			limit = { check_variable = { i < 4 }
					  check_variable = { SCB_trade_num^i = 2 } }
			set_variable = { SCB_Hainan_trade_equip_amount^i = SCB_QYC_light_amount_temp }
			round_variable = SCB_Hainan_trade_equip_amount^i
		}
		else_if = {
			limit = { check_variable = { i > 3 }
					  check_variable = { i < 6 }
					  check_variable = { SCB_trade_num^i = 2 } }
			set_variable = { SCB_Hainan_trade_equip_amount^i = SCB_QYC_heavy_amount_temp }
		}
		else_if = {
			limit = { check_variable = { SCB_trade_num^i = 1 } }
			set_variable = { SCB_Hainan_trade_equip_amount^i = 0 }
		}
	}
	set_temp_variable = { SCB_temp_amount_sum_l = 0 }
	set_temp_variable = { SCB_temp_amount_sum_h = 0 }
	for_each_loop = {
		array = SCB_Hainan_trade_equip_amount
		index = i
		if = {
			limit = { check_variable = { i < 4 } }
			add_to_temp_variable = { SCB_temp_amount_sum_l = SCB_Hainan_trade_equip_amount^i }
		}
		else_if = {
			limit = { check_variable = { i > 3 }
					  check_variable = { i < 6 } }
			add_to_temp_variable = { SCB_temp_amount_sum_h = SCB_Hainan_trade_equip_amount^i }
		}
	}
	subtract_from_temp_variable = { SCB_temp_amount_sum_l = QYC.SCB_QYC_light_amount }
	subtract_from_temp_variable = { SCB_temp_amount_sum_h = QYC.SCB_QYC_heavy_amount }
	for_each_loop = {
		array = SCB_Hainan_trade_equip_amount
		index = i
		if = {
			limit = { check_variable = { i < 4 }
					  check_variable = { SCB_trade_num^i = 2 } }
			subtract_from_variable = { SCB_Hainan_trade_equip_amount^i = SCB_temp_amount_sum_l }
			set_temp_variable = { SCB_temp_amount_sum_l = 0 }
		}
		else_if = {
			limit = { check_variable = { i > 3 }
					  check_variable = { i < 6 }
					  check_variable = { SCB_trade_num^i = 2 } }
			subtract_from_variable = { SCB_Hainan_trade_equip_amount^i = SCB_temp_amount_sum_h }
			set_temp_variable = { SCB_temp_amount_sum_h = 0 }
		}
	}
}
