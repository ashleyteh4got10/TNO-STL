OMS_chemical_weapons_gui_stockpile_change = {
	add_to_variable = { OMS_chemical_weapons_gui_current_stockpile = OMS_chemical_weapons_gui_adding_to_stockpile }
	if = {
		limit = { check_variable = {OMS_chemical_weapons_gui_adding_to_stockpile > 0}}
		custom_effect_tooltip = OMS_chemical_weapons_gui_stockpile_increase_tt
	}
	else_if = {
		limit = { check_variable = {OMS_chemical_weapons_gui_adding_to_stockpile < 0}}
		custom_effect_tooltip = OMS_chemical_weapons_gui_stockpile_decrease_tt
	}
	clamp_variable = {
		var = OMS_chemical_weapons_gui_current_stockpile
		min = 0
		max = OMS_chemical_weapons_gui_stockpile_limit
	}
}

OMS_chemical_weapons_gui_stability_change = {
	add_to_variable = { OMS_chemical_weapons_gui_chemical_stability = OMS_chemical_weapons_gui_chemical_stability_change }
	add_to_variable = { OMS_chemical_weapons_gui_biologic_stability = OMS_chemical_weapons_gui_biologic_stability_change }

	else_if = {
		limit = { check_variable = { OMS_chemical_weapons_gui_biologic_stability_change < 0 } }
		custom_effect_tooltip = OMS_chemical_weapons_gui_biologic_stability_decrease_tt

	clamp_variable = {
		var = OMS_chemical_weapons_gui_chemical_stability

	}
