calc_politics_powers_in_CPC = {
	if = {
		limit = { check_variable = { NCCPC_temp_pct_change_chosen = 0 } }
		calc_pies_in_CPC = yes
	}
	else_if = {
		limit = { check_variable = { NCCPC_temp_pct_change_chosen > 0 } }
		set_temp_variable = { NCCPC_temp_pct_change = 100 }
		subtract_from_temp_variable = { NCCPC_temp_pct_change = TER_pol_str_real_percent^CPC_NCCPC_choose_num }
		divide_temp_variable = { NCCPC_temp_pct_change = NCCPC_temp_pct_change_chosen }
		add_to_variable = { TER_pol_str_real_percent^CPC_NCCPC_choose_num = NCCPC_temp_pct_change }
		calc_pies_in_CPC = yes
	}
	else_if = {
		limit = { check_variable = { NCCPC_temp_pct_change_chosen < 0 } }
		set_temp_variable = { NCCPC_temp_pct_change = TER_pol_str_real_percent^CPC_NCCPC_choose_num }
		divide_temp_variable = { NCCPC_temp_pct_change = NCCPC_temp_pct_change_chosen }
		add_to_variable = { TER_pol_str_real_percent^CPC_NCCPC_choose_num = NCCPC_temp_pct_change }
		calc_pies_in_CPC = yes
	}
}

	add_to_temp_variable = { NCCPC_end_1 = TER_pol_str_real_percent^1 }
	set_variable = { TER_PS_printed_percent^1 = NCCPC_st_1 }

	multiply_variable = { TER_PS_printed_percent^1 = 1000 }
	add_to_variable = { TER_PS_printed_percent^1 = NCCPC_end_1 }

		add_to_array = { NCCPC_Line_graph_vbars = label }

		set_temp_variable = { offset = k }
		subtract_from_temp_variable = { offset = min }
		divide_temp_variable = { offset = range }
		multiply_temp_variable = { offset = -1.5 }
		add_to_array = { NCCPC_Line_graph_vbar_offsets = offset }

		add_to_temp_variable = { k = unit }
	}
	for_loop_effect = {
		end = 16
		value = i
		subtract_from_temp_variable = { NCCPC_Line_graph_vbars^i = divisor}
	}
	#实际生成图像部分
	for_loop_effect = {
		start = 0
		end = 15
		value = i

		set_temp_variable = { j = i }
		add_to_temp_variable = { j = 1 }

		set_temp_variable = { vj = NCCPC_Line_graph_values^j }
		set_temp_variable = { vi = NCCPC_Line_graph_values^i }
		subtract_from_temp_variable = { vj = min }
		subtract_from_temp_variable = { vi = min }
		divide_temp_variable = { vj = range }
		divide_temp_variable = { vi = range }

		divide_temp_variable = { vj = 2 }
		divide_temp_variable = { vi = 2 }
		round_temp_variable = vi
		round_temp_variable = vj
		multiply_temp_variable = { vj = 2 }
		multiply_temp_variable = { vi = 2 }

		set_temp_variable = { delta = vj }
		subtract_from_temp_variable = { delta = vi }
		divide_temp_variable = { delta = 2 }
		round_temp_variable = delta
		add_to_temp_variable = { delta = 50 }

		set_temp_variable = { offset = vi }
		multiply_temp_variable = { offset = -1.5 }

		add_to_array = { NCCPC_Line_graph_offsets = offset }
		add_to_array = { NCCPC_Line_graph_deltas = delta }
	}
	
}

NCCPC_yearly_clear_the_budget = {
	set_variable = { NCCPC_yearly_factor_temp = NCCPC_current_party_budget }
	divide_variable = { NCCPC_yearly_factor_temp = NCCPC_party_budget_limit }
	if = {
		limit = { check_variable = { NCCPC_yearly_factor_temp > 0.5 } }
		set_variable = { NCCPC_buff_factor = -0.01 }
		NCCPC_buff_yearly_calc = yes
	}
	else_if = {
		limit = { OR = { check_variable = { NCCPC_yearly_factor_temp = 0.5 } 
				  		 AND = { check_variable = { NCCPC_yearly_factor_temp < 0.5 }
				  		 		 check_variable = { NCCPC_yearly_factor_temp > 0.1 } } } }
		set_variable = { NCCPC_buff_factor = 0.05 }
		NCCPC_buff_yearly_calc = yes
	}
	else_if = {
		limit = { OR = { check_variable = { NCCPC_yearly_factor_temp < 0.1 }
				  		 check_variable = { NCCPC_yearly_factor_temp = 0.1 } }
				  OR = { check_variable = { NCCPC_yearly_factor_temp > 0 }
				  		 check_variable = { NCCPC_yearly_factor_temp = 0 } } }
		set_variable = { NCCPC_buff_factor = -0.02 }
		NCCPC_buff_yearly_calc = yes
	}
	else_if = {
		limit = { check_variable = { NCCPC_yearly_factor_temp < 0 } }
		set_variable = { NCCPC_buff_factor = -0.07 }
		NCCPC_buff_yearly_calc = yes
	}
	set_temp_variable = { NCCPC_next_party_budget_temp = NCCPC_next_party_budget }
	divide_temp_variable = { NCCPC_next_party_budget_temp = 1000 } #转单位为M
	set_temp_variable = { debt_temp = NCCPC_next_party_budget_temp }
	econ_national_debt_change = yes
	set_temp_variable = { NCCPC_current_party_budget_temp = NCCPC_next_party_budget }
	divide_temp_variable = { NCCPC_current_party_budget_temp = 1000 } #转单位为M
	set_temp_variable = { money_reserves_temp = NCCPC_current_party_budget_temp }
	econ_money_reserves_change_raw_money = yes
	set_variable = { NCCPC_current_party_budget = NCCPC_next_party_budget }
	set_variable = { NCCPC_party_budget_limit = NCCPC_current_party_budget }
	clr_country_flag = NCCPC_buget_confirmed
	TER_NCCPC_budget_fire_alerts = yes
}
