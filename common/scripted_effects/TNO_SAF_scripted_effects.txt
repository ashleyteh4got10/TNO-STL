SAF_Chaos_change = {
	set_temp_variable = { SAF_new_Chaos = SAF_Chaos }
	add_to_temp_variable = { SAF_new_Chaos = SAF_Chaos_temp }

	if = {
		limit = {
			has_country_flag = SAF_Chaos_locked_at_100
		}
		set_temp_variable = { SAF_Chaos_temp = 0 }
	} 
	else_if = {
		limit = {
			# if value is below the chaos bound
			check_variable = { SAF_Chaos_temp < 0 }
			check_variable = { SAF_Chaos < SAF_Chaos_bound }
		}
		divide_temp_variable = { SAF_Chaos_temp = 3 }
		add_to_variable = { SAF_Chaos = SAF_Chaos_temp }
	} 
	else_if = {
		limit = {
			# if new value would be below the soft bound
			check_variable = { SAF_Chaos_temp < 0 }
			check_variable = { SAF_new_Chaos < SAF_Chaos_bound }
		}
		# take midpoint between bound and new val
		set_temp_variable = { SAF_new_Chaos = SAF_Chaos_bound }
		add_to_temp_variable = { SAF_new_Chaos = SAF_Chaos }
		add_to_temp_variable = { SAF_new_Chaos = SAF_Chaos_temp }
		divide_temp_variable = { SAF_new_Chaos = 2 }

		set_temp_variable = { SAF_Chaos_temp = SAF_new_Chaos }
		subtract_from_temp_variable = { SAF_Chaos_temp = SAF_Chaos }
		set_variable = { SAF_Chaos = SAF_new_Chaos }
	} 
	else = {
		add_to_variable = { SAF_Chaos = SAF_Chaos_temp }
	}

	set_temp_variable = { SAF_Chaos_temp_abs = SAF_Chaos_temp }
	if = {
		limit = {
			check_variable = { SAF_Chaos_temp < 0 }
		}
		multiply_temp_variable = { SAF_Chaos_temp_abs = -1 }
	}
	custom_effect_tooltip = SAF_IA_chaos_change_tt

	clamp_variable = {
		var = SAF_Chaos
		min = 0
		max = 100
	}
	
	if = {
		limit = {
			check_variable = { SAF_Chaos > 74.999 }
		}
		
	}
	SAF = { # Just want to 100% make sure this is tied to SAF
		if = {
			limit = {
				has_country_flag = SAF_Div_Horse_Flag
			}
			set_country_leader_portrait = {
				portrait = "gfx/leaders/SAF/Portrait_South_Africa_Sir_de_Viller_Graaf_75_chaos_oguri.png"
			}
		}
		else = {
			set_country_leader_portrait = {
				portrait = "gfx/leaders/SAF/Portrait_South_Africa_Sir_de_Viller_Graaf_75_chaos.png"
			}
		}
	}

	SAF_refresh_chaos_idea = yes
}

			modifier = SAF_Chaos_modifier
		}
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = SAF_Chaos_modifier_2 } }
		remove_dynamic_modifier = {
			modifier = SAF_Chaos_modifier_2
		}
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = SAF_Chaos_modifier_3 } }
		remove_dynamic_modifier = {
			modifier = SAF_Chaos_modifier_3

SAF_refresh_chaos_idea = {
	hidden_effect = {
		set_variable = { SAF_Chaos_stability = 0 }
		set_variable = { SAF_Chaos_war_support = 0 }
		set_variable = { SAF_Chaos_misc_costs = 0 }
		set_variable = { SAF_Chaos_recruitable_pop = 0 }
		set_variable = { SAF_Chaos_construction_speed = 0 }
		set_variable = { SAF_Chaos_gdp_growth = 0 }
		set_variable = { SAF_Chaos_research_speed = 0 }
		if = {
			limit = {
				check_variable = { SAF_Chaos < 10 }
			}
			set_variable = { SAF_Chaos_stability = 0.2 }
			SAF_recalculate_power_tick_change_sub_20 = yes
			SAF_remove_chaos_modifiers = yes
			add_dynamic_modifier = {
				modifier = SAF_Chaos_modifier
			}
		} else_if = {
			limit = {
				check_variable = { SAF_Chaos < 20 }
			}
			set_variable = { SAF_Chaos_stability = 0.1 }
			SAF_recalculate_power_tick_change_sub_20 = yes
			SAF_remove_chaos_modifiers = yes
			add_dynamic_modifier = {
				modifier = SAF_Chaos_modifier
			}
		} else_if = {
			limit = {
				check_variable = { SAF_Chaos < 40 }
			}
			# no effects outside of power tick
			SAF_recalculate_power_tick_change_above_20 = yes
			SAF_remove_chaos_modifiers = yes
			add_dynamic_modifier = {
				modifier = SAF_Chaos_modifier_2
			}
		} else_if = {
			limit = {
				check_variable = { SAF_Chaos < 60 }
			}
			set_variable = { SAF_Chaos_stability = -0.05 }
			set_variable = { SAF_Chaos_war_support = -0.05 }
			set_variable = { SAF_Chaos_misc_costs = 0.06 }
			SAF_recalculate_power_tick_change_above_20 = yes
			SAF_remove_chaos_modifiers = yes
			add_dynamic_modifier = {
				modifier = SAF_Chaos_modifier_3
			}
		} else_if = {
			limit = {
				check_variable = { SAF_Chaos < 80 }
			}
			set_variable = { SAF_Chaos_stability = -0.075 }
			set_variable = { SAF_Chaos_war_support = -0.075 }
			set_variable = { SAF_Chaos_misc_costs = 0.096 }
			set_variable = { SAF_Chaos_recruitable_pop = -0.05 }
			set_variable = { SAF_Chaos_construction_speed = -0.05 }
			set_variable = { SAF_Chaos_gdp_growth = -0.2 }
			SAF_recalculate_power_tick_change_above_20 = yes
			SAF_remove_chaos_modifiers = yes
			add_dynamic_modifier = {
				modifier = SAF_Chaos_modifier_4
			}
		} else_if = {
			limit = {
				check_variable = { SAF_Chaos < 90 }
			}
			set_variable = { SAF_Chaos_stability = -0.1 }
			set_variable = { SAF_Chaos_war_support = -0.1 }
			set_variable = { SAF_Chaos_misc_costs = 0.12 }
			set_variable = { SAF_Chaos_recruitable_pop = -0.1 }
			set_variable = { SAF_Chaos_construction_speed = -0.1 }

SAF_recalculate_power_tick_change_sub_20 = {
	set_variable = { SAF_Chaos_power_increase_tick = SAF_Chaos }

	add_to_variable = { SAF_Chaos_power_increase_tick = -0.20 }
}

SAF_recalculate_power_tick_change_above_20 = {

	if = {
		limit = {
			surrender_progress < 0.01
		}
		set_variable = { THIS.SAW_surrender_buff = 0 }
	}
	else_if = {

SAF_pakt_anger_change = {
	if = {
		limit = {
			NOT = { has_country_flag = SAF_ended_pakt_trade }
		}
		set_temp_variable = { new_displeasure = SAF_Pakt_Displeasure }
		add_to_temp_variable = { new_displeasure = SAF_Pakt_Displeasure_temp } #temp var bullshit for stuff below
		clamp_temp_variable = {
			var = new_displeasure
			min = 0 
			max = 1
		}
		set_variable = { SAF_Pakt_Displeasure = new_displeasure }

		set_temp_variable = { SAF_Pakt_Displeasure_temp_abs = SAF_Pakt_Displeasure_temp }
		if = {
			limit = {
				check_variable = { SAF_Pakt_Displeasure_temp < 0 }
			}
			multiply_temp_variable = { SAF_Pakt_Displeasure_temp_abs = -1 }
		}
		custom_effect_tooltip = TNO_SAF_Pakt_displeasure_tt
		
		if = { #check if we need to cut off trade
			limit = {
				check_variable = { new_displeasure = 1 }
			}
			SAF = {
				country_event = { id = SAF.173 days = 1 }
			}
		} else_if = { #percentage chance for cutoff
			limit = {
				check_variable = { new_displeasure > 0.59 } #60 or up
			}
			set_temp_variable = { p_cutoff = 5 }
			if = {
				limit = {
					check_variable = { new_displeasure > 0.89 } #90 or up
				}
				set_temp_variable = { p_cutoff = 40 }
			} else_if = {
				limit = {
					check_variable = { new_displeasure > 0.79 } #80 or up
				}
				set_temp_variable = { p_cutoff = 20 }
			} else_if = {
				limit = {
					check_variable = { new_displeasure > 0.69 } #70 or up
				}
				set_temp_variable = { p_cutoff = 10 }
			}
			set_temp_variable = { p_not_cutoff = 100 }
			subtract_from_temp_variable = { p_not_cutoff = p_cutoff }
			random_list = {
				p_cutoff = {
					SAF = {
						country_event = { id = SAF.173 days = 1 }
					}
				}
				p_not_cutoff = { }
			}
		}

		#calculate which angry hitler to use
		set_temp_variable = { SAF_Pakt_Displeasure_frame_temp = SAF_Pakt_Displeasure }
		multiply_temp_variable = { SAF_Pakt_Displeasure_frame_temp = 10 }
		add_to_temp_variable = { SAF_Pakt_Displeasure_frame_temp = 0.5 } #approximate floor
		round_temp_variable = SAF_Pakt_Displeasure_frame_temp
		set_variable = { SAF_Pakt_Displeasure_frame = SAF_Pakt_Displeasure_frame_temp }
		clamp_variable = {
			var = SAF_Pakt_Displeasure_frame
			min = 1 
			max = 10
		}
		SAF_mark_rhodes_gui_dirty = yes
	}
}

## give this scripted effect a postive number for SAF_society_progressivism_temp please!
## moves progressivism towards 100 (status quo) by the amount of points specified in the temp variable 
## if the value of the temp variable would overshoot 100, (i.e progressivism is 105 and you give it 15)
## the change will be modified to not overshoot (i.e the change would become 5 instead of 15 to exactly hit 100)
SAF_Society_progressivism_middle_change = {
	# calculate diff between current and 100
	set_temp_variable = { SAF_diff_from_status_quo = SAF_society_progressivism }
	subtract_from_temp_variable = { SAF_diff_from_status_quo = 100 }

	set_temp_variable = { SAF_diff_from_status_quo_abs = SAF_diff_from_status_quo }
	if = { 
		limit = {
			check_variable = { SAF_diff_from_status_quo < 0 }
		}
		
		multiply_temp_variable = { SAF_diff_from_status_quo_abs = -1 }
	}

	if = {
		limit = {
			check_variable = { SAF_society_progressivism_temp > SAF_diff_from_status_quo_abs }
		}
		set_temp_variable = { SAF_society_progressivism_temp = SAF_diff_from_status_quo_abs }
	}

	if = {
		limit = {
			check_variable = { SAF_society_progressivism > 100 }
		}
		multiply_temp_variable = { SAF_society_progressivism_temp = -1 }
	}

	custom_effect_tooltip = SAF_society_status_quo_tt

	hidden_effect = {
		SAF_Society_progressivism_change = yes
	}
}

SAF_endings_determine_event = {
	if = {
		limit = {
			check_variable = {
				var = SAF_society_progressivism
				compare = greater_than_or_equals
				value = 161
			}
		}
		log = "[GetDateText]: [Root.GetName]: SAF ENDING: VERY PROGRESSIVE"
		set_country_flag = SAF_ending_very_progressive
		country_event = { id = HDV_SAF.200 hours = 6 }
	} else_if = {
		limit = {
			check_variable = {
				var = SAF_society_progressivism
				compare = greater_than_or_equals
				value = 121
			}
			check_variable = { SAF_society_progressivism < 161 }
		}
		log = "[GetDateText]: [Root.GetName]: SAF ENDING: PROGRESSIVE"
		set_country_flag = SAF_ending_progressive
		country_event = { id = HDV_SAF.209 hours = 6 }
	} else_if = {
		limit = {
			check_variable = {
				var = SAF_society_progressivism
				compare = greater_than_or_equals
				value = 81
			}
			check_variable = { SAF_society_progressivism < 121 }
		}
		log = "[GetDateText]: [Root.GetName]: SAF ENDING: STATUS QUO"
		set_country_flag = SAF_ending_status_quo
		country_event = { id = HDV_SAF.218 hours = 6 }
	} else_if = {
		limit = {
			check_variable = {
				var = SAF_society_progressivism
				compare = greater_than_or_equals
				value = 40
			}
			check_variable = { SAF_society_progressivism < 81 }
		}
		log = "[GetDateText]: [Root.GetName]: SAF ENDING: CONSERVATIVE"
		set_country_flag = SAF_ending_conservative
		country_event = { id = HDV_SAF.225 hours = 6 }
	} else_if = {
		limit = {
			check_variable = { SAF_society_progressivism < 40 }

		set_country_flag = SAF_ending_very_conservative
		country_event = { id = HDV_SAF.234 hours = 6 }
	}
}
